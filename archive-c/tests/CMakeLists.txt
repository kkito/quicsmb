if(QUICSMB_USE_VCPKG)
  find_package(cmocka CONFIG REQUIRED)
  set(QS_CMOCKA_TARGET cmocka::cmocka)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CMOCKA REQUIRED cmocka)
endif()

function(add_qs_test name src)
    add_executable(${name} ${src})
    if(QUICSMB_USE_VCPKG)
      target_link_libraries(${name} PRIVATE ${QS_CMOCKA_TARGET} quicsmb_core)
    else()
      target_include_directories(${name} PRIVATE ${CMOCKA_INCLUDE_DIRS})
      target_link_libraries(${name} PRIVATE ${CMOCKA_LIBRARIES} quicsmb_core)
    endif()
    target_include_directories(${name} PRIVATE ${CMAKE_SOURCE_DIR}/include)
    add_test(NAME ${name} COMMAND ${name})
endfunction()

add_qs_test(test_smb unit/test_smb.c)
add_qs_test(test_config unit/test_config.c)
add_qs_test(test_quic_stub unit/test_quic_stub.c)
