// Package chunk 做文件偏移与块号的纯映射、切分与校验和。
package chunk

import "hash/fnv"

const DefaultBlockSize = 512 << 10

type ChunkID struct {
	File  string
	Index uint64
}

func ValidBlockSize(n int) bool {
	switch n {
	case 256 << 10, 512 << 10, 1 << 20, 4 << 20:
		return true
	}
	return false
}

func Count(sizeBytes int64, blockSize int) int {
	if sizeBytes <= 0 {
		return 0
	}
	return int((sizeBytes + int64(blockSize) - 1) / int64(blockSize))
}

func Split(sizeBytes int64, blockSize int) []ChunkID {
	n := Count(sizeBytes, blockSize)
	out := make([]ChunkID, 0, n)
	for i := 0; i < n; i++ {
		out = append(out, ChunkID{Index: uint64(i)})
	}
	return out
}

func IndexAt(offset int64, blockSize int) uint64 {
	return uint64(offset / int64(blockSize))
}

func RangeOf(index uint64, blockSize int) (int64, int) {
	return int64(index) * int64(blockSize), blockSize
}

func Checksum(b []byte) uint64 {
	h := fnv.New64a()
	_, _ = h.Write(b)
	return h.Sum64()
}
```

// Package cache 是内存 LRU 块缓存：配额淘汰、命中率、访问热度。
package cache

import (
	"container/list"
	"sort"
	"sync"

	"quicsmb/internal/chunk"
)

type Key = chunk.ChunkID

type entry struct {
	key  Key
	data []byte
	heat uint64
}

type Cache struct {
	mu        sync.Mutex
	cap       int64
	used      int64
	ll        *list.List
	items     map[Key]*list.Element
	hits      uint64
	total     uint64
}

func New(capBytes int64, blockSize int) *Cache {
	_ = blockSize
	return &Cache{cap: capBytes, ll: list.New(), items: map[Key]*list.Element{}}
}

func (c *Cache) Put(k Key, data []byte) {
	c.mu.Lock()
	defer c.mu.Unlock()
	if el, ok := c.items[k]; ok {
		c.used += int64(len(data)) - int64(len(el.Value.(*entry).data))
		el.Value.(*entry).data = append([]byte(nil), data...)
		c.ll.MoveToFront(el)
		return
	}
	cp := append([]byte(nil), data...)
	el := c.ll.PushFront(&entry{key: k, data: cp})
	c.items[k] = el
	c.used += int64(len(cp))
	for c.used > c.cap {
		if !c.evictOldestLocked() {
			break
		}
	}
}

func (c *Cache) Get(k Key) ([]byte, bool) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.total++
	el, ok := c.items[k]
	if !ok {
		return nil, false
	}
	c.hits++
	e := el.Value.(*entry)
	e.heat++
	c.ll.MoveToFront(el)
	return append([]byte(nil), e.data...), true
}

func (c *Cache) EvictOldest() bool {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.evictOldestLocked()
}

func (c *Cache) evictOldestLocked() bool {
	back := c.ll.Back()
	if back == nil {
		return false
	}
	e := back.Value.(*entry)
	delete(c.items, e.key)
	c.used -= int64(len(e.data))
	c.ll.Remove(back)
	return true
}

func (c *Cache) Usage() (int64, int) {
	c.mu.Lock()
	defer c.mu.Unlock()
	return c.used, len(c.items)
}

func (c *Cache) HitRate() float64 {
	c.mu.Lock()
	defer c.mu.Unlock()
	if c.total == 0 {
		return 0
	}
	return float64(c.hits) / float64(c.total)
}

func (c *Cache) Heat(k Key) uint64 {
	c.mu.Lock()
	defer c.mu.Unlock()
	if el, ok := c.items[k]; ok {
		return el.Value.(*entry).heat
	}
	return 0
}

func (c *Cache) HeatTop(n int) []Key {
	c.mu.Lock()
	defer c.mu.Unlock()
	all := make([]*entry, 0, len(c.items))
	for _, el := range c.items {
		all = append(all, el.Value.(*entry))
	}
	sort.Slice(all, func(i, j int) bool { return all[i].heat > all[j].heat })
	if n > len(all) {
		n = len(all)
	}
	out := make([]Key, 0, n)
	for i := 0; i < n; i++ {
		out = append(out, all[i].key)
	}
	return out
}
```

// Package metastore 持久化脏块元数据：脏时间、校验和、失败计数、块大小一致性。
package metastore

import (
	"encoding/json"
	"fmt"
	"os"
	"sync"
)

type DirtyEntry struct {
	File      string `json:"file"`
	Index     uint64 `json:"index"`
	DirtyUnix int64  `json:"dirty_unix"`
	Sum       uint64 `json:"sum"`
	Failures  int    `json:"failures"`
}

type diskForm struct {
	BlockSize int          `json:"block_size"`
	Entries   []DirtyEntry `json:"entries"`
}

type Store struct {
	mu        sync.Mutex
	path      string
	blockSize int
	items     map[string]*DirtyEntry
}

func key(file string, index uint64) string {
	return fmt.Sprintf("%s\x00%d", file, index)
}

func New(path string, blockSize int) *Store {
	return &Store{path: path, blockSize: blockSize, items: map[string]*DirtyEntry{}}
}

func Load(path string, blockSize int) (*Store, error) {
	s := New(path, blockSize)
	b, err := os.ReadFile(path)
	if err != nil {
		if os.IsNotExist(err) {
			return s, nil
		}
		return nil, err
	}
	var f diskForm
	if err := json.Unmarshal(b, &f); err != nil {
		return nil, err
	}
	if f.BlockSize != 0 && f.BlockSize != blockSize {
		return nil, fmt.Errorf("block size changed: meta=%d want=%d (清空缓存后重启)", f.BlockSize, blockSize)
	}
	for _, e := range f.Entries {
		cp := e
		s.items[key(e.File, e.Index)] = &cp
	}
	return s, nil
}

func (s *Store) MarkDirty(file string, index uint64, sum uint64, nowUnix int64) {
	s.mu.Lock()
	defer s.mu.Unlock()
	k := key(file, index)
	if e, ok := s.items[k]; ok {
		e.Sum = sum
		return
	}
	s.items[k] = &DirtyEntry{File: file, Index: index, DirtyUnix: nowUnix, Sum: sum}
}

func (s *Store) MarkClean(file string, index uint64) {
	s.mu.Lock()
	defer s.mu.Unlock()
	delete(s.items, key(file, index))
}

func (s *Store) BumpFailure(file string, index uint64) {
	s.mu.Lock()
	defer s.mu.Unlock()
	if e, ok := s.items[key(file, index)]; ok {
		e.Failures++
	}
}

func (s *Store) List() []DirtyEntry {
	s.mu.Lock()
	defer s.mu.Unlock()
	out := make([]DirtyEntry, 0, len(s.items))
	for _, e := range s.items {
		out = append(out, *e)
	}
	return out
}

func (s *Store) Save() error {
	s.mu.Lock()
	defer s.mu.Unlock()
	f := diskForm{BlockSize: s.blockSize}
	for _, e := range s.items {
		f.Entries = append(f.Entries, *e)
	}
	b, err := json.Marshal(f)
	if err != nil {
		return err
	}
	tmp := s.path + ".tmp"
	if err := os.WriteFile(tmp, b, 0600); err != nil {
		return err
	}
	return os.Rename(tmp, s.path)
}
```

// Package auth 做预共享 Key 的包认证：HMAC 校验，失败无差别拒绝（黑洞）。
package auth

import (
	"crypto/hmac"
	"crypto/rand"
	"crypto/sha256"
)

const nonceLen = 16
const macLen = 32

func Nonce16() ([16]byte, error) {
	var n [16]byte
	_, err := rand.Read(n[:])
	return n, err
}

func Seal(key, nonce []byte, payload []byte) []byte {
	if len(nonce) != nonceLen {
		var n [nonceLen]byte
		copy(n[:], nonce)
		nonce = n[:]
	}
	m := hmac.New(sha256.New, key)
	m.Write(nonce)
	m.Write(payload)
	out := make([]byte, 0, nonceLen+macLen+len(payload))
	out = append(out, nonce...)
	out = append(out, m.Sum(nil)...)
	out = append(out, payload...)
	return out
}

func Open(key []byte, msg []byte) ([]byte, bool) {
	if len(msg) < nonceLen+macLen {
		return nil, false
	}
	nonce := msg[:nonceLen]
	want := msg[nonceLen : nonceLen+macLen]
	payload := msg[nonceLen+macLen:]
	m := hmac.New(sha256.New, key)
	m.Write(nonce)
	m.Write(payload)
	if !hmac.Equal(m.Sum(nil), want) {
		return nil, false
	}
	return payload, true
}
```

// Package writeback 是回写调度的纯判定：满延时 + 低峰窗口 + 失败上限。
package writeback

import "sort"

type Clock func() int64

type Policy struct {
	DelaySec    int64
	WinStartMin int
	WinEndMin   int
	MaxFailures int
}

type DirtyEntryView struct {
	DirtyUnix int64
	Failures  int
}

func DefaultPolicy() Policy {
	return Policy{DelaySec: 72 * 3600, WinStartMin: 120, WinEndMin: 180, MaxFailures: 10}
}

func inWindow(nowUnix int64, p Policy) bool {
	m := int((nowUnix / 60) % (24 * 60))
	if p.WinStartMin <= p.WinEndMin {
		return m >= p.WinStartMin && m < p.WinEndMin
	}
	return m >= p.WinStartMin || m < p.WinEndMin
}

func Due(e DirtyEntryView, p Policy, now Clock) bool {
	t := now()
	if e.Failures >= p.MaxFailures {
		return false
	}
	if t-e.DirtyUnix < p.DelaySec {
		return false
	}
	return inWindow(t, p)
}

func SelectDue(all []DirtyEntryView, p Policy, now Clock, limit int) []int {
	var idx []int
	for i, e := range all {
		if Due(e, p, now) {
			idx = append(idx, i)
		}
	}
	sort.Slice(idx, func(a, b int) bool {
		return all[idx[a]].DirtyUnix < all[idx[b]].DirtyUnix
	})
	if limit > 0 && len(idx) > limit {
		idx = idx[:limit]
	}
	return idx
}
